## Installation

### CLI

```package-exec
solidui-cli@latest add calendar
```

### Manual

Install the following dependencies:

```package-install
@corvu/calendar @date-fns/tz date-fns lucide-solid
```

Copy and paste the following code into your project.

```file="~/registry/ui/calendar.tsx" showLineNumbers

```

## Usage

```tsx showLineNumbers
import { createSignal } from "solid-js";
import { Calendar } from "~/components/ui/calendar";

const [date, setDate] = createSignal<Date>();
```

```tsx showLineNumbers
<Calendar
  mode="single"
  selected={date()}
  onSelect={setDate}
  class="rounded-lg border"
/>
```

See the [Corvu Calendar documentation](https://corvu.dev/docs/primitives/calendar) for primitive-level options and behavior.

### Date Picker

You can use the `<Calendar>` component to build a date picker by combining it with `<Popover>` and `<Button>` components. See the Date Picker examples below.

### Custom Cell Size

You can customize the size of calendar cells using the `--cell-size` CSS variable. You can also make it responsive by using breakpoint-specific values:

```tsx showLineNumbers
<Calendar
  mode="single"
  value={date()}
  onValueChange={setDate}
  class="rounded-lg border [--cell-size:--spacing(11)] md:[--cell-size:--spacing(12)]"
/>
```

### Week Numbers

Display ISO week numbers alongside the calendar using the `weekNumbers` prop:

```tsx showLineNumbers
<Calendar mode="single" value={date()} onValueChange={setDate} weekNumbers />
```

### Custom Cells

Use the `customCell` prop to add custom content to each day cell. This is useful for displaying metadata like prices, events, or availability indicators:

```tsx showLineNumbers
import { Show } from "solid-js";
import { Calendar, type CustomCellProps } from "~/components/ui/calendar";

const renderPriceCell = (props: CustomCellProps) => {
  const price = getPriceForDate(props.date);

  return (
    <Show when={price !== null && !props.isOutsideMonth}>
      <span
        class={`text-[0.65rem] ${props.isSelected ? "text-primary-foreground/80" : "text-muted-foreground"}`}
      >
        ${price}
      </span>
    </Show>
  );
};

<Calendar
  mode="single"
  value={date()}
  onValueChange={setDate}
  customCell={renderPriceCell}
  class="[--cell-size:--spacing(11)]"
/>;
```

### Month/Year Selection

Enable dropdown selection for month and year using the `monthYearSelection` prop:

```tsx showLineNumbers
<Calendar
  mode="single"
  value={date()}
  onValueChange={setDate}
  monthYearSelection
  startYear={1990}
  endYear={2030}
/>
```

### Combining with Time Inputs

You can combine the calendar with time inputs to create a date-time picker:

```tsx showLineNumbers
import { createSignal } from "solid-js";
import { Calendar } from "~/components/ui/calendar";
import { Input } from "~/components/ui/input";

const [date, setDate] = createSignal<Date | null>(null);
const [time, setTime] = createSignal("10:30");

<div>
  <Calendar mode="single" value={date()} onValueChange={setDate} />
  <Input
    type="time"
    value={time()}
    onInput={(e) => setTime(e.currentTarget.value)}
  />
</div>;
```

### Styling Disabled Dates

You can style disabled dates differently using the `disabled` prop and CSS selectors:

```tsx showLineNumbers
import { Calendar } from "~/components/ui/calendar";

const bookedDates = [new Date(2024, 0, 15), new Date(2024, 0, 20)];

const isBooked = (day: Date) => {
  return bookedDates.some(
    (booked) => booked.toDateString() === day.toDateString(),
  );
};

<Calendar
  mode="single"
  disabled={isBooked}
  class="**:data-[disabled=true]:line-through **:data-[disabled=true]:opacity-30"
/>;
```

## API Reference

### Calendar

The Calendar wrapper follows the [Corvu Calendar API reference](https://corvu.dev/docs/primitives/calendar#api-reference) and adds the DayPicker-compatible props shown below.

| Prop              | Type                                                             | Default   | Description                              |
| ----------------- | ---------------------------------------------------------------- | --------- | ---------------------------------------- |
| `mode`            | `"single" \| "multiple" \| "range"`                              | -         | Selection mode.                          |
| `selected`        | `Date \| Date[] \| { from: Date; to?: Date }`                    | -         | Controlled selected value.               |
| `onSelect`        | `(selected, triggerDate, modifiers, event) => void`              | -         | Called after selection changes.          |
| `month`           | `Date`                                                           | -         | Controlled month to display.             |
| `onMonthChange`   | `(month: Date) => void`                                          | -         | Called when the displayed month changes. |
| `captionLayout`   | `"label" \| "dropdown" \| "dropdown-months" \| "dropdown-years"` | `"label"` | Caption presentation.                    |
| `numberOfMonths`  | `number`                                                         | `1`       | Number of visible months.                |
| `showOutsideDays` | `boolean`                                                        | `true`    | Whether adjacent-month days are visible. |
| `showWeekNumber`  | `boolean`                                                        | `false`   | Whether to show ISO week numbers.        |
| `disabled`        | `CalendarMatcher \| CalendarMatcher[]`                           | -         | Dates that cannot be selected.           |
| `class`           | `string`                                                         | -         | Additional classes on the root element.  |

For primitive props, keyboard behavior, and accessibility details, see the [Corvu Calendar API reference](https://corvu.dev/docs/primitives/calendar#api-reference).
